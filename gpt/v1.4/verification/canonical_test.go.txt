func canonicalSetup(t *testing.T)(*TxService,*xorm.Engine){
 e:=setup(t)
 for _,q:=range []string{
  "ALTER TABLE idempotency_records ADD COLUMN result_version INTEGER NOT NULL DEFAULT 0",
  "INSERT INTO books(id,name,owner_uid) VALUES(1,'test book',1)",
  "INSERT INTO book_members(book_id,uid,role) VALUES(1,1,1)",
  "INSERT INTO categories(id,book_id,cat_type,name) VALUES(10,1,2,'expense'),(20,1,2,'other expense'),(11,1,1,'income')",
 } {if _,err:=e.Exec(q);err!=nil{t.Fatal(err)}}
 return NewTxService(e,&VersionRepo{}),e
}
func transferRequest(key string)CreateRequest{return CreateRequest{ClientUUID:key,TxType:3,AccountID:1,ToAccountID:2,Amount:1000,TxTime:100}}
func TestUnified_TransferReplayReturnsSameIDsAndVersion(t *testing.T){
 s,e:=canonicalSetup(t);r:=transferRequest("replay")
 first,err:=s.Create(context.Background(),1,1,r);if err!=nil{t.Fatal(err)}
 replay,err:=s.Create(context.Background(),1,1,r);if err!=nil{t.Fatal(err)}
 if !replay.Replayed||first.Replayed||!reflect.DeepEqual(first.TransactionIDs,replay.TransactionIDs)||first.Version!=replay.Version{t.Fatal("replay differs",first,replay)}
 if scalar(t,e,"SELECT COUNT(*) FROM transactions")!=2||scalar(t,e,"SELECT version FROM book_versions WHERE book_id=1")!=1{t.Fatal("replay mutated business")}
 if scalar(t,e,"SELECT COUNT(*) FROM accounts WHERE sync_version=1")!=2||scalar(t,e,"SELECT COUNT(*) FROM transactions WHERE tx_time=100 AND sync_version=1")!=2{t.Fatal("partial version/time")}
 if scalar(t,e,"SELECT COUNT(*) FROM transactions WHERE related_id>0")!=2{t.Fatal("unpaired transfer")}
}

func TestUnified_AllAssetLiabilityDirectionsPreserveNetAssets(t *testing.T){
 for _,pair:=range [][2]int{{1,1},{1,10},{10,1},{10,10}}{
  t.Run(string(rune('A'+pair[0]))+string(rune('A'+pair[1])),func(t *testing.T){
   s,e:=canonicalSetup(t)
   if _,err:=e.Exec("UPDATE accounts SET category=?,balance=10000 WHERE id=1",pair[0]);err!=nil{t.Fatal(err)}
   if _,err:=e.Exec("UPDATE accounts SET category=?,balance=10000 WHERE id=2",pair[1]);err!=nil{t.Fatal(err)}
   sign:=func(cat int)int64{if cat>=10{return -1};return 1}
   before:=int64(10000)*sign(pair[0])+int64(10000)*sign(pair[1])
   if _,err:=s.Create(context.Background(),1,1,transferRequest("directions"));err!=nil{t.Fatal(err)}
   after:=scalar(t,e,"SELECT balance FROM accounts WHERE id=1")*sign(pair[0])+scalar(t,e,"SELECT balance FROM accounts WHERE id=2")*sign(pair[1])
   if after!=before{t.Fatal("net assets changed",pair,before,after)}
  })
 }
}

func TestUnified_ConcurrentRetriesReturnOneResult(t *testing.T){
 s,e:=canonicalSetup(t);var wg sync.WaitGroup;out:=make(chan *CreateResult,16);errs:=make(chan error,16)
 for i:=0;i<16;i++{wg.Add(1);go func(){defer wg.Done();r,err:=s.Create(context.Background(),1,1,transferRequest("parallel"));out<-r;errs<-err}()}
 wg.Wait();close(out);close(errs)
 for err:=range errs{if err!=nil{t.Fatal(err)}}
 var want []int64
 for r:=range out{if want==nil{want=r.TransactionIDs};if !reflect.DeepEqual(want,r.TransactionIDs)||r.Version!=1{t.Fatal("different replay result")}}
 if scalar(t,e,"SELECT COUNT(*) FROM transactions")!=2||scalar(t,e,"SELECT COUNT(*) FROM idempotency_records")!=1{t.Fatal("duplicate transfer")}
}
