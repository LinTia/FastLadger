func setup(t *testing.T) *xorm.Engine {
 t.Helper()
 path:=filepath.Join(t.TempDir(),"case.db")
 e,err:=xorm.NewEngine("sqlite3","file:"+path+"?_journal_mode=WAL&_foreign_keys=on&_busy_timeout=100&_synchronous=NORMAL")
 if err!=nil {t.Fatal(err)}
 e.SetMaxOpenConns(1)
 raw,err:=os.ReadFile("../../schema.sql");if err!=nil {t.Fatal(err)}
 if _,err=e.Exec(string(raw));err!=nil {t.Fatal(err)}
 database.Engine=e
 t.Cleanup(func(){e.Close()})
 for _,q:=range []string{
  "INSERT INTO accounts(id,book_id,category,name,balance,sync_version) VALUES(1,1,1,'cash',100000,0)",
  "INSERT INTO accounts(id,book_id,category,name,balance,sync_version) VALUES(2,1,1,'wallet',0,0)",
 } {if _,err=e.Exec(q);err!=nil{t.Fatal(err)}}
 return e
}
func scalar(t *testing.T,e *xorm.Engine,q string) int64 {
 t.Helper();var n int64
 if err:=e.DB().QueryRow(q).Scan(&n);err!=nil{t.Fatal(err)}
 return n
}
func observation(t *testing.T,v any){t.Helper();b,_:=json.Marshal(v);t.Log("OBSERVATION",string(b))}

func TestV14_ReceiptInsertFailureAcknowledgedAsSuccess(t *testing.T){
 e:=setup(t)
 _,err:=e.Exec("CREATE TRIGGER fail_receipt BEFORE INSERT ON idempotency_records BEGIN SELECT RAISE(ABORT,'synthetic receipt failure'); END")
 if err!=nil{t.Fatal(err)}
 err=Transfer(1,1,1,2,1000,"receipt-fault")
 n:=scalar(t,e,"SELECT COUNT(*) FROM transactions")
 if err!=nil||n!=0{t.Fatalf("expected documented false acknowledgement; err=%v rows=%d",err,n)}
 observation(t,map[string]any{"error":nil,"transactions":n,"balance":scalar(t,e,"SELECT balance FROM accounts WHERE id=1")})
}

func TestV14_RollbackThenRetryRunsOnce(t *testing.T){
 e:=setup(t)
 _,err:=e.Exec("CREATE TRIGGER fail_second_leg BEFORE INSERT ON transactions WHEN NEW.account_id=2 BEGIN SELECT RAISE(ABORT,'synthetic second-leg failure'); END")
 if err!=nil{t.Fatal(err)}
 if err=Transfer(1,1,1,2,1000,"rollback-key");err==nil{t.Fatal("injected business error must surface")}
 if scalar(t,e,"SELECT COUNT(*) FROM idempotency_records")!=0||scalar(t,e,"SELECT COUNT(*) FROM transactions")!=0||scalar(t,e,"SELECT COUNT(*) FROM book_versions")!=0{t.Fatal("rollback left rows")}
 if scalar(t,e,"SELECT balance FROM accounts WHERE id=1")!=100000{t.Fatal("rollback changed balance")}
 if _,err=e.Exec("DROP TRIGGER fail_second_leg");err!=nil{t.Fatal(err)}
 if err=Transfer(1,1,1,2,1000,"rollback-key");err!=nil{t.Fatal(err)}
 if err=Transfer(1,1,1,2,1000,"rollback-key");err!=nil{t.Fatal(err)}
 if scalar(t,e,"SELECT COUNT(*) FROM transactions")!=2||scalar(t,e,"SELECT version FROM book_versions WHERE book_id=1")!=1{t.Fatal("retry duplicated work")}
 observation(t,map[string]any{"rollback_clears_receipt":true,"retry_and_replay_transactions":2,"version":1})
}

func TestV14_ConcurrentSameKeyWithSingleConnection(t *testing.T){
 e:=setup(t);var wg sync.WaitGroup;errs:=make(chan error,16)

 wg.Wait();close(errs);for err:=range errs{if err!=nil{t.Fatal(err)}}

}
