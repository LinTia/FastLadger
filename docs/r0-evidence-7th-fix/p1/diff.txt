+/**

+/**
+ * 递归收集文件；符号链接一律跳过，不跟随。
+ * @param {string} dir 目录绝对路径。
+ * @param {{files: string[], links: string[]}} out 输出累加器。
+ * @throws {Error} 目录不可读时抛出。

+  let entries;
+  try {
+    entries = fs.readdirSync(dir, { withFileTypes: true });
+  } catch (err) {
+    throw new Error(`无法读取目录 ${displayPath(dir)}：${err.message}`);

+    if (entry.isSymbolicLink()) {
+      out.links.push(abs);
+      continue;
+    }
+    if (entry.isDirectory()) {
+      walk(abs, out);
+      continue;
+    }
+    if (entry.isFile()) out.files.push(abs);
+  }

+ * @param {string} dest 目标目录。
+ * @throws {Error} 组合非法时抛出。

+function validatePaths(source, dest) {

+  if (dest.startsWith(source + path.sep) || source.startsWith(dest + path.sep)) {
+    throw new Error('源目录与目标目录不能互相嵌套');
+  }
+  if (dest === REPO_ROOT || REPO_ROOT.startsWith(dest + path.sep)) {
+    throw new Error('目标目录不能是仓库根或其上级目录');
+  }
+  if (dest === path.join(REPO_ROOT, 'artifacts')) {
+    throw new Error('目标目录不能是 artifacts 根目录（清空操作会删除全部证据）');
+  }
+}

+  }

+  const { source, dest } = options;
+  try {

+    process.stderr.write(`[prepare][ERROR] ${err.message}\n`);
+    return EXIT_ERROR;
+  }
+
+  let sourceStat;
+  try {
+    sourceStat = fs.statSync(source);
+  } catch (err) {
+    process.stderr.write(`[prepare][ERROR] 源目录不存在或不可访问：${displayPath(source)}\n`);
+    return EXIT_ERROR;
+  }
+  if (!sourceStat.isDirectory()) {
+    process.stderr.write(`[prepare][ERROR] 源路径不是目录：${displayPath(source)}\n`);
+    return EXIT_ERROR;
+  }
+
+  const tree = { files: [], links: [] };
+  try {
+    walk(source, tree);
+  } catch (err) {
+    process.stderr.write(`[prepare][ERROR] ${err.message}\n`);
+    return EXIT_ERROR;
+  }
+
+  /** @type {{rel: string, reason: string}[]} */
+  const excluded = [];
+  /** @type {string[]} */
+  const collected = [];
+
+  try {
+    // 先清空，保证上传目录只含本轮白名单结果，杜绝上一轮残留。

+      fs.copyFileSync(abs, target);
+      collected.push(rel);
+    }
+  } catch (err) {
+    process.stderr.write(`[prepare][ERROR] 生成白名单副本失败：${err.message}\n`);
+    return EXIT_ERROR;
+  }
+
+  collected.sort();
+  excluded.sort((a, b) => a.rel.localeCompare(b.rel));
+
+  const lines = [
+    `[prepare] 源目录：${displayPath(source)}`,
+    `[prepare] 目标目录：${displayPath(dest)}（运行前已清空）`,
+    `[prepare] 白名单：${[...ALLOWED_EXT].join(' ')}`,
+    `[prepare] 排除清单（共 ${excluded.length}，仅文件名与原因，不含内容）：`,
+  ];
+  for (const item of excluded) lines.push(`  - ${item.rel}  [${item.reason}]`);
+  if (excluded.length === 0) lines.push('  （无）');
+  lines.push(`[prepare] 收集清单（共 ${collected.length}，仅文件名，不含内容）：`);
+  for (const rel of collected) lines.push(`  - ${rel}`);
+  if (collected.length === 0) lines.push('  （无）');
+  lines.push(`[prepare] 符号链接跳过：${tree.links.length}`);
+  lines.push(
+    `[prepare] 汇总：扫描=${tree.files.length} 收集=${collected.length} ` +
+    `排除=${excluded.length} 符号链接跳过=${tree.links.length}`,
+  );
+  lines.push('[prepare] 结论：OK（输出目录已就绪，仍需经 scan-artifacts.js 扫描后方可上传）');
+  process.stdout.write(`${lines.join('\n')}\n`);
+  return EXIT_OK;
+}
+

+  for (let i = 0; i < argv.length; i += 1) {
+    const arg = argv[i];
+    if (arg === '--dir') {
+      const value = argv[i + 1];
+      if (!value || value.startsWith('--')) throw new Error('参数 --dir 缺少取值');
+      dir = path.resolve(process.cwd(), value);
+      i += 1;
+      continue;
+    }
+    if (arg.startsWith('--')) throw new Error(`未知参数：${arg}`);
+    if (positionalSeen) throw new Error('只能指定一个待扫描目录');
+    dir = path.resolve(process.cwd(), arg);
+    positionalSeen = true;
+  }
+  return { dir, help: false };

+/**
+ * 统计某条规则在文本中的命中次数，使用独立正则副本避免 lastIndex 污染。
+ * @param {string} text 待扫描文本。
+ * @param {RegExp} re 带 g 标志的规则。
+ * @return {number} 命中次数。
+ */
+function countMatches(text, re) {
+  const copy = new RegExp(re.source, re.flags);
+  let count = 0;
+  let match;
+  while ((match = copy.exec(text)) !== null) {
+    count += 1;
+    if (copy.lastIndex === match.index) copy.lastIndex += 1;
+  }
+  return count;
+}

+/**
+ * 检查文件名是否命中黑名单。
+ * @param {string} rel 相对路径。
+ * @return {string[]} 命中类型列表。

+function fileNameFindings(rel) {
+  const base = path.basename(rel);
+  const lower = base.toLowerCase();
+  const ext = path.extname(lower);
+  const types = [];
+
+  if (BLACKLIST_EXT.has(ext)) {
+    types.push(ext === '.zip' && lower.includes('trace') ? 'TRACE_ARCHIVE' : 'BLACKLIST_EXT');
+  }
+  if (lower === SERVER_LOG_NAME) types.push('SERVER_LOG');
+  if (SENSITIVE_NAME_RE.test(base)) types.push('SENSITIVE_FILENAME');
+  return types;
+}

+/**
+ * 通过 NUL 字节判定二进制内容。
+ * @param {Buffer} buf 文件内容。
+ * @return {boolean} 是否为二进制。
+ */
+function looksBinary(buf) {
+  const limit = Math.min(buf.length, 4096);
+  for (let i = 0; i < limit; i += 1) {
+    if (buf[i] === 0) return true;
+  }
+  return false;
+}

+/**
+ * 递归遍历目录；符号链接单独记录，不跟随。
+ * @param {string} dir 目录绝对路径。
+ * @param {{files: string[], symlinks: string[]}} out 输出累加器。
+ * @throws {Error} 目录不可读时抛出。
+ */
+function walk(dir, out) {
+  let entries;
+  try {
+    entries = fs.readdirSync(dir, { withFileTypes: true });
+  } catch (err) {
+    throw new Error(`无法读取目录 ${displayPath(dir)}：${err.message}`);
+  }
+  for (const entry of entries) {
+    const abs = path.join(dir, entry.name);
+    if (entry.isSymbolicLink()) {
+      out.symlinks.push(abs);
+      continue;
+    }
+    if (entry.isDirectory()) {
+      walk(abs, out);
+      continue;
+    }
+    if (entry.isFile()) out.files.push(abs);
+  }
+}

+/**
+ * 程序入口。
+ * @param {string[]} argv process.argv.slice(2)。
+ * @return {number} 退出码。

+function main(argv) {
+  let options;
+  try {
+    options = parseArgs(argv);
+  } catch (err) {
+    process.stderr.write(`[scan][ERROR] ${err.message}\n`);
+    return EXIT_ERROR;
+  }
+  if (options.help) {
+    printHelp();
+    return EXIT_CLEAN;
+  }
+
+  const targetDir = options.dir;
+  let dirStat;
+  try {
+    dirStat = fs.statSync(targetDir);
+  } catch (err) {
+    process.stderr.write(`[scan][ERROR] 待扫描目录不存在或不可访问：${displayPath(targetDir)}\n`);
+    return EXIT_ERROR;
+  }
+  if (!dirStat.isDirectory()) {
+    process.stderr.write(`[scan][ERROR] 待扫描路径不是目录：${displayPath(targetDir)}\n`);
+    return EXIT_ERROR;
+  }
+
+  /** @type {Map<string, {file: string, type: string, count: number}>} */
+  const findings = new Map();
+  /**
+   * 累加一条命中，只记录文件名、类型与数量。
+   * @param {string} file 相对文件路径。
+   * @param {string} type 命中类型。
+   * @param {number} count 命中数量。
+   */
+  function addFinding(file, type, count) {
+    const key = `${file}|${type}`;
+    const prev = findings.get(key);
+    if (prev) prev.count += count;
+    else findings.set(key, { file, type, count });
+  }
+
+  const tree = { files: [], symlinks: [] };
+  let textScanned = 0;
+  let binarySkipped = 0;
+
+  try {
+    walk(targetDir, tree);
+    for (const abs of tree.files) {
+      const rel = path.relative(targetDir, abs).split(path.sep).join('/');
+      for (const type of fileNameFindings(rel)) addFinding(rel, type, 1);
+
+      const ext = path.extname(abs).toLowerCase();
+      if (BINARY_EXT.has(ext)) {
+        binarySkipped += 1;
+        continue;
+      }
+      const stat = fs.statSync(abs);
+      if (stat.size > MAX_TEXT_BYTES) {
+        throw new Error(`文件超过可扫描上限 ${MAX_TEXT_BYTES} 字节：${rel}`);
+      }
+      const buf = fs.readFileSync(abs);
+      if (looksBinary(buf)) {
+        binarySkipped += 1;
+        continue;
+      }
+      const text = buf.toString('utf8');
+      textScanned += 1;
+      for (const rule of TEXT_RULES) {
+        const count = countMatches(text, rule.re);
+        if (count > 0) addFinding(rel, rule.type, count);
+      }
+    }
+    for (const link of tree.symlinks) {
+      const rel = path.relative(targetDir, link).split(path.sep).join('/');
+      addFinding(rel, 'SYMLINK_ENTRY', 1);
+    }
+  } catch (err) {
+    process.stderr.write(`[scan][ERROR] 扫描过程异常：${err.message}\n`);
+    return EXIT_ERROR;
+  }
+
+  const rows = [...findings.values()].sort(
+    (a, b) => a.file.localeCompare(b.file) || a.type.localeCompare(b.type),
+  );
+  const totalHits = rows.reduce((sum, row) => sum + row.count, 0);
+
+  const lines = [
+    `[scan] 待上传目录：${displayPath(targetDir)}`,
+    `[scan] 文件总数=${tree.files.length} 文本已扫描=${textScanned} ` +
+      `二进制跳过内容扫描=${binarySkipped} 符号链接=${tree.symlinks.length}`,
+    `[scan] 命中明细（仅文件路径 + 命中类型 + 命中数量，不输出匹配值）：`,
+  ];
+  for (const row of rows) lines.push(`[HIT] ${row.file} | ${row.type} | ${row.count}`);
+  if (rows.length === 0) lines.push('[HIT] （无） | — | 0');
+  lines.push(`[scan] 命中文件/类型组合=${rows.length} 命中总数=${totalHits}`);
+
+  if (totalHits > 0) {
+    lines.push('[scan] 结论：BLOCKED —— 禁止上传，请先清除上述命中项');
+    process.stdout.write(`${lines.join('\n')}\n`);
+    return EXIT_BLOCKED;
+  }
+  if (tree.files.length === 0) {
+    lines.push('[scan][WARN] 目录为空；上传 step 的 if-no-files-found=error 仍会使 job 失败');
+  }
+  lines.push('[scan] 结论：CLEAN —— 允许上传');
+  process.stdout.write(`${lines.join('\n')}\n`);
+  return EXIT_CLEAN;
+}
+
+process.exitCode = main(process.argv.slice(2));
