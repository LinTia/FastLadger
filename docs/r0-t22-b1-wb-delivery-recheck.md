```python
#!/usr/bin/env python3
# V-10 矩阵验收校验（只读）。用法：python3 v10_check.py docs/r0-status-matrix-13x8.md
import sys, re, os, glob

STATES = ['normal','loading','empty','error','success','permission','offline','conflict']
ok, bad = [], []

def band(lines, a, b):
    s = next((i for i,l in enumerate(lines) if l.startswith(a)), None)
    e = next((i for i,l in enumerate(lines) if l.startswith(b)), None)
    if s is None: raise SystemExit(f'缺章节 {a}')
    return lines[s:(e if e else len(lines))]

def expand(seg):
    m = re.search(r'\{([^{}]*)\}', seg)
    if not m: return [seg]
    pre, suf = seg[:m.start()], seg[m.end():]
    out = []
    for o in m.group(1).split(','): out += expand(pre + o + suf)
    return out

def main(path):
    L = open(path, encoding='utf-8').read().splitlines()
    text = '\n'.join(L)
    m = band(L, '## 1.', '## 2.')
    rows = []
    for l in m:
        mm = re.match(r'^\| ([0-9]+) \|', l)
        if not mm: continue
        f = [x.strip() for x in l.split('|')[1:-1]]
        if len(f) != 10: bad.append(f'§1 行列数 {len(f)} ≠ 10')
        rows.append((int(f[0]), f[1], f[2:10]))
    if len(rows) != 13: bad.append(f'§1 行数 {len(rows)} ≠ 13')
    else: ok.append('§1 行数=13')
    cells = {}
    for no, page, sts in rows:
        for st, v in zip(STATES, sts):
            k = 'N/A' if v.startswith('N/A') else '有' if v.startswith('有') else '缺' if v.startswith('缺') else None
            if k is None: bad.append(f'{page}/{st} 非法值「{v}」')
            if k == '有' and ('缺' in v or 'N/A' in v): bad.append(f'{page}/{st} 混值「{v}」')
            cells[(no, st)] = (k, v)
    cnt = {k: sum(1 for v in cells.values() if v[0] == k) for k in ('有','缺','N/A')}
    if sum(cnt.values()) != 104: bad.append(f'总格 {sum(cnt.values())} ≠ 104')
    else: ok.append(f'总格=104（有 {cnt["有"]} / 缺 {cnt["缺"]} / N/A {cnt["N/A"]}）')
    na = {k: v for k, v in cells.items() if v[0] == 'N/A'}
    ledger = band(L, '## 3.', '## 4.')
    names = {no: page for no, page, _ in rows}
    mapped = {}
    for l in ledger:
        if not re.match(r'^\| [0-9]+ \|', l): continue
        f = [x.strip() for x in l.split('|')[1:-1]]
        no = next((n for n,p in names.items() if p == f[1]), None)
        if no is None: bad.append(f'§3 页面名「{f[1]}」与 §1 不一致'); continue
        mapped[(no, f[2])] = f[3]
    if set(mapped) != set(na): bad.append('N/A 台账未闭合')
    else: ok.append(f'N/A 台账闭合 {len(mapped)}/{len(na)}')
    for k in na:
        if k in mapped and mapped[k] not in na[k][1]: bad.append(f'{k} 格内码与台账码不一致')
    ev = band(L, '## 4.', '## 5.')
    eids = {}
    for l in ev:
        mm = re.match(r'^\| (E[0-9]+) \|', l)
        if mm:
            f = [x.strip() for x in l.split('|')[1:-1]]
            eids[f[0]] = f
    used = []
    for k, (kk, v) in cells.items():
        if kk == '有':
            mm = re.search(r'(E[0-9]+)', v)
            if not mm: bad.append(f'{k} 记「有」但未引 E#')
            else: used.append(mm.group(1))
    if set(used) != set(eids): bad.append('E 台账与「有」格未一一对应')
    else: ok.append(f'E 台账一一对应 {len(eids)} 格')
    base = 'docs/r0-screenshots/'
    miss = []
    for eid, f in eids.items():
        paths = f[5].replace('`','').replace(' + ','+')
        exps = []
        for seg in paths.split('+'): exps += expand(seg)
        if not any(os.path.isfile(base + p) for p in exps): miss.append(eid)
    if miss: bad.append(f'E 路径不可解析：{miss}')
    else: ok.append('E 台账路径全部可解析')
    if not re.search(r'视口齐备|三视口', text): bad.append('缺「视口齐备」统计口径')
    off = [v for (n, st), (k, v) in cells.items() if k == 'N/A' and st == 'offline']
    if off and all(('R0-E-15' not in v and 'R0-E-25' not in v) for v in off):
        bad.append('offline N/A 格均未注明已知缺口编号')
    if 'R0-E-15' not in text and 'R0-E-25' not in text: bad.append('全文无 R0-E-15 / R0-E-25 缺口登记')
    if '基准' not in text or 'SHA' not in text: bad.append('证据台账缺「代码基准 SHA」')
    if '数据前置' not in text: bad.append('证据台账缺「数据前置」')
    if '96' not in text: bad.append('缺「适用格 96（12×8）」分母')
    print('通过项：')
    for x in ok: print('  OK  ' + x)
    print('未通过项：')
    for x in bad: print('  FAIL ' + x)
    print(f'结论：{"PASS" if not bad else "FAIL"}（未通过 {len(bad)} 项）')
    return 1 if bad else 0

if __name__ == '__main__':
    sys.exit(main(sys.argv[1] if len(sys.argv) > 1 else 'docs/r0-status-matrix-13x8.md'))
```
